        // ============ ANÁLISE ============
        function analyzeCode() {
            const code = document.getElementById('input-code').value;
            if (!code.trim()) {
                updateStatus('❌ Cole um código primeiro!', 'error');
                return;
            }

            state.originalCode = code;
            state.analysis = {
                functions: [],
                variables: [],
                imports: [],
                styles: [],
                html: ''
            };

            // Extrair HTML
            const htmlMatch = code.match(/]*>[\s\S]*<\/body>/i);
            if (htmlMatch) state.analysis.html = htmlMatch[0];

            // Extrair CSS
            const styleMatches = code.match(/]*>[\s\S]*?<\/style>/gi) || [];
            state.analysis.styles = styleMatches;

            // Extrair funções
            const funcRegex = /function\s+(\w+)\s*\(([^)]*)\)\s*{/g;
            let match;
            while ((match = funcRegex.exec(code)) !== null) {
                state.analysis.functions.push({
                    name: match[1],
                    params: match[2],
                    type: 'function'
                });
            }

            // Extrair variáveis globais
            const varRegex = /(?:let|const|var)\s+(\w+)\s*=/g;
            while ((match = varRegex.exec(code)) !== null) {
                state.analysis.variables.push(match[1]);
            }

            // Extrair imports
            const importRegex = /(?:import|require)\s+(?:.*\s+)?from\s+['"]([^'"]+)['"]/g;
            while ((match = importRegex.exec(code)) !== null) {
                state.analysis.imports.push(match[1]);
            }

            displayAnalysis();
        }

        function displayAnalysis() {
            let output = `📊 ANÁLISE DO CÓDIGO

    `;
            output += `🔍 Funções encontradas: ${state.analysis.functions.length}
    `;
            state.analysis.functions.forEach(f => {
                output += `  • ${f.name}(${f.params})
    `;
            });

            output += `
    📦 Variáveis globais: ${state.analysis.variables.length}
    `;
            state.analysis.variables.slice(0, 10).forEach(v => {
                output += `  • ${v}
    `;
            });

            output += `
    📥 Imports/Requires: ${state.analysis.imports.length}
    `;
            state.analysis.imports.forEach(i => {
                output += `  • ${i}
    `;
            });

            output += `
    🎨 Estilos CSS: ${state.analysis.styles.length}
    `;
            output += `📄 HTML: ${state.analysis.html.length} caracteres
    `;

            document.getElementById('analysis-output').innerHTML = output;
            updateStatus('✓ Análise concluída!', 'success');
        }

        // ============ SEPARAÇÃO DE CÓDIGO ============
        function separateCode() {
            const code = document.getElementById('input-code').value;
            if (!code.trim()) {
                updateStatus('❌ Cole um código primeiro!', 'error');
                return;
            }

            state.modules = {};
            updateProgress(0);

            // 1. Extrair HTML
            const htmlMatch = code.match(/]*>[\s\S]*<\/body>/i);
            state.modules['index.html'] = generateHTML(code, htmlMatch ? htmlMatch[0] : '');
            updateProgress(20);

            // 2. Extrair CSS
            const styles = code.match(/]*>([\s\S]*?)<\/style>/gi) || [];
            let cssContent = '';
            styles.forEach(s => {
                const match = s.match(/]*>([\s\S]*?)<\/style>/i);
                if (match) cssContent += match[1] + '\n\n';
            });
            if (cssContent) {
                state.modules['css/styles.css'] = cssContent;
            }
            updateProgress(40);

            // 3. Extrair JavaScript em funções
            const scriptMatch = code.match(/]*>([\s\S]*?)<\/script>/i);
            if (scriptMatch) {
                const jsCode = scriptMatch[1];
                
                // Core
                state.modules['js/core.js'] = extractCore(jsCode);
                updateProgress(50);

                // Funções de Editor
                state.modules['js/editor.js'] = extractModule(jsCode, 'editor');
                updateProgress(60);

                // Funções de Mapper
                state.modules['js/mapper.js'] = extractModule(jsCode, 'mapper');
                updateProgress(70);

                // Funções de Sincronização
                state.modules['js/sync.js'] = extractSyncFunctions(jsCode);
                updateProgress(80);

                // Utilitários
                state.modules['js/utils.js'] = extractUtils(jsCode);
                updateProgress(90);
            }

            // 4. Criar arquivo de configuração
            state.modules['package.json'] = generatePackageJson();
            updateProgress(95);

            // 5. Criar README
            state.modules['README.md'] = generateReadme();
            updateProgress(100);

            displayModules();
            updateStatus('✓ Código separado com sucesso!', 'success');
        }

        function extractCore(jsCode) {
            let core = `// ============ ESTADO GLOBAL ============\n`;
            core += `let state = {\n`;
            core += `    files: {},\n`;
            core += `    currentFile: null,\n`;
            core += `    analysis: {},\n`;
            core += `    config: {}\n`;
            core += `};\n\n`;

            // Extrair variáveis globais
            const varRegex = /(?:let|const|var)\s+(\w+)\s*=\s*([^;]+);/g;
            let match;
            while ((match = varRegex.exec(jsCode)) !== null) {
                if (!match[1].startsWith('_')) {
                    core += `// ${match[1]}\n`;
                }
            }

            core += `\n// ============ INICIALIZAÇÃO ============\n`;
            core += `function init() {\n`;
            core += `    loadFromStorage();\n`;
            core += `    renderUI();\n`;
            core += `}\n\n`;

            core += `function loadFromStorage() {\n`;
            core += `    const saved = localStorage.getItem('iara-state');\n`;
            core += `    if (saved) state = JSON.parse(saved);\n`;
            core += `}\n\n`;

            core += `function saveToStorage() {\n`;
            core += `    localStorage.setItem('iara-state', JSON.stringify(state));\n`;
            core += `}\n\n`;

            return core;
        }

        function extractModule(jsCode, moduleName) {
            let module = `// ============ MÓDULO: ${moduleName.toUpperCase()} ============\n\n`;

            const funcRegex = new RegExp(`function\\s+\\w*${moduleName}\\w*\\s*\\([^)]*\\)\\s*{[^}]*}`, 'gi');
            const matches = jsCode.match(funcRegex) || [];

            if (matches.length === 0) {
                module += `// Funções relacionadas a ${moduleName}\n\n`;
                module += `function ${moduleName}Init() {\n`;
                module += `    console.log('${moduleName} iniciado');\n`;
                module += `}\n\n`;
            } else {
                matches.forEach(func => {
                    module += func + '\n\n';
                });
            }

            return module;
        }

        function extractSyncFunctions(jsCode) {

            sync += `        return false;\n`;

            sync += `    try {\n`;
            sync += `        // Atualizar estado\n`;
            sync += `        state.currentFile = data.fileName || 'mapper-output';\n`;
            sync += `        state.files[state.currentFile] = data.content || '';\n`;
            sync += `        saveToStorage();\n\n`;

            sync += `        renderFileTree();\n\n`;

            sync += `        return true;\n`;
            sync += `    } catch (error) {\n`;
            sync += `        console.error('Erro ao enviar:', error);\n`;
            sync += `        return false;\n`;
            sync += `    }\n`;
            sync += `}\n\n`;

            sync += ` * Função que recebe dados do Editor para o Mapper\n`;
            sync += ` */\n`;
            sync += `function sendEditorToMapper(fileContent) {\n`;
            sync += `    return {\n`;
            sync += `        fileName: state.currentFile,\n`;
            sync += `        content: fileContent,\n`;
            sync += `        timestamp: new Date().toISOString()\n`;
            sync += `    };\n`;
            sync += `}\n\n`;

            sync += `// Event Listeners para sincronização\n`;
            sync += `document.addEventListener('DOMContentLoaded', () => {\n`;
            sync += `    // Sincronizar quando arquivo é salvo\n`;
            sync += `    document.addEventListener('file-saved', (e) => {\n`;
            sync += `        console.log('Arquivo salvo:', e.detail);\n`;
            sync += `    });\n\n`;
            sync += `    // Sincronizar quando mapper envia dados\n`;
            sync += `    document.addEventListener('mapper-send', (e) => {\n`;
            sync += `        sendMapperToEditor(e.detail);\n`;
            sync += `    });\n`;
            sync += `});\n`;

            return sync;
        }

        function extractUtils(jsCode) {
            let utils = `// ============ UTILITÁRIOS ============\n\n`;
            utils += `function formatDate(date) {\n`;
            utils += `    return new Date(date).toLocaleString('pt-BR');\n`;
            utils += `}\n\n`;

            utils += `function generateId() {\n`;
            utils += `    return '_' + Math.random().toString(36).substr(2, 9);\n`;
            utils += `}\n\n`;

            utils += `function debounce(func, wait) {\n`;
            utils += `    let timeout;\n`;
            utils += `    return function executedFunction(...args) {\n`;
            utils += `        const later = () => {\n`;
            utils += `            clearTimeout(timeout);\n`;
            utils += `            func(...args);\n`;
            utils += `        };\n`;
            utils += `        clearTimeout(timeout);\n`;
            utils += `        timeout = setTimeout(later, wait);\n`;
            utils += `    };\n`;
            utils += `}\n\n`;

            utils += `function throttle(func, limit) {\n`;
            utils += `    let inThrottle;\n`;
            utils += `    return function(...args) {\n`;
            utils += `        if (!inThrottle) {\n`;
            utils += `            func.apply(this, args);\n`;
            utils += `            inThrottle = true;\n`;
            utils += `            setTimeout(() => inThrottle = false, limit);\n`;
            utils += `        }\n`;
            utils += `    };\n`;
            utils += `}\n`;

            return utils;
        }
